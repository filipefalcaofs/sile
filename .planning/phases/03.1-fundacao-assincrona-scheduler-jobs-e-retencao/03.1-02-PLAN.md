       O job define `public int $tries = 3;`, que TEM PRECEDÊNCIA sobre `--tries` do worker. Com `queue:work --once`, cada execução faz UMA tentativa; ao falhar, o job é LIBERADO de volta à fila com o `backoff` (delay), e só vai para `failed_jobs` quando `attempts() >= $tries` (3ª tentativa). Portanto NÃO usar `--tries=1` (não tem efeito) — é preciso esgotar as 3 tentativas, avançando o relógio (`$this->travel`) além do backoff entre as execuções para o job voltar a ficar disponível:
       ```php
       config(['queue.default' => 'database']);

       ImportRedesimJob::dispatch(base_path('nao-existe.json'));

       // backoff = [30, 60, 120]; avançar o relógio torna o job disponível de novo.
       $this->artisan('queue:work', ['--once' => true]); // tentativa 1 → release (delay 30s)
       $this->travel(31)->seconds();
       $this->artisan('queue:work', ['--once' => true]); // tentativa 2 → release (delay 60s)
       $this->travel(61)->seconds();
       $this->artisan('queue:work', ['--once' => true]); // tentativa 3 → attempts >= tries → failed_jobs

       $this->assertSame(1, \Illuminate\Support\Facades\DB::table('failed_jobs')->count());
       $this->assertDatabaseHas('activity_log', [
           'log_name' => 'empresas',
           'event' => 'importacao-redesim',
           'result' => 'falha',
       ]);
       ```
       ÁREA A VALIDAR (anotar no SUMMARY): `queue:work` real sobre sqlite `:memory:` depende de a conexão ser compartilhada no mesmo processo do teste — funciona com `--once` (Laravel reusa o PDO único do `:memory:`), e `$this->travel()` afeta tanto o `now()` do release quanto a query de disponibilidade da fila (`DatabaseQueue` usa `Carbon::now()`). Se, na execução, o `:memory:` não compartilhar a conexão, trocar para o banco de arquivo de teste padrão NÃO é necessário — primeiro confirmar o comportamento rodando o teste.

    ```php
    namespace App\Console\Commands;

    use App\Jobs\ImportCnaeJob;
    use App\Services\CnaeImportService;
    use App\Support\Audit\AuditService;
    use Illuminate\Console\Command;
    use RuntimeException;

    class ImportCnaeCommand extends Command
    {
        protected $signature = 'cnae:importar {arquivo : Caminho do CSV oficial CNAE-Subclasses 2.3} {--queue : Processa em fila (job com retry/timeout) em vez de síncrono}';

        protected $description = 'Reimporta a estrutura oficial de CNAEs a partir de um CSV (HU-011) — caminho de reimportação oficial';

        public function handle(CnaeImportService $service, AuditService $audit): int
        {
            $path = (string) $this->argument('arquivo');

            if (! is_file($path)) {
                $this->error("Arquivo não encontrado: {$path}");

                return self::FAILURE;
            }

            if ($this->option('queue')) {
                ImportCnaeJob::dispatch($path);
                $this->info('Importação CNAE enfileirada (job ImportCnaeJob).');

                return self::SUCCESS;
            }

            try {
                $report = $service->import($path);
            } catch (RuntimeException $e) {
                $this->error('Falha na importação CNAE: '.$e->getMessage());

                return self::FAILURE;
            }

            $audit->log(
                logName: 'cnaes',
                event: 'importacao-oficial',
                description: 'Reimportação da estrutura oficial CNAE-Subclasses 2.3 via comando',
                properties: $report,
                rulesVersion: 'cnae-subclasses-2.3',
            );

            $this->info('Importação CNAE concluída.');
            $this->line("Lidos: {$report['lidos']}");
            $this->line("Importados: {$report['importados']}");
            $this->line("Atualizados: {$report['atualizados']}");

            if ($report['rejeitados'] !== []) {
                $this->newLine();
                $this->warn('Rejeitados:');
                foreach ($report['rejeitados'] as $motivo) {
                    $this->line("  - {$motivo}");
                }
            }

            return self::SUCCESS;
        }
    }
    ```
