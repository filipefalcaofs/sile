    ```php
    abstract class PostgisTestCase extends TestCase
    {
        use RefreshDatabase;

        protected function createApplication()
        {
            $app = parent::createApplication();
            // Aponta a conexão default para o Postgres+PostGIS ANTES do migrate (setUpTraits).
            $app['config']->set('database.default', 'pgsql_testing');
            return $app;
        }

        // RefreshDatabase chama este hook ANTES de migrar: garante a extensão.
        protected function beforeRefreshingDatabase(): void
        {
            try {
                DB::connection('pgsql_testing')->statement('CREATE EXTENSION IF NOT EXISTS postgis');
            } catch (\Throwable $e) {
                $this->markTestSkipped('PostGIS indisponível para testes espaciais: '.$e->getMessage());
            }
        }
    }
    ```
